# Per-site knowledge and procedures: how a site behaves, and what to do about it.
# One line per site; every key is optional:
#
#     <host>  [keep=N|all] [min=SIZE] [id=REGEX] [also=HOSTS] [serial] [native]
#             [probe=HOSTS] [gdl=KEY=VAL,...] [key=PARAM] [playlist=SEG]
#             [postid=REGEX] [posturl=URL] [itemurl=URL] [tpl=TEMPLATE]
#             [folder=REGEX] [cookiedrop=NAMES] [engine=NAME] [author=NAME] [dest=NAME]
#             [rewrite=REGEX=>URL]
#     *       [strip=REGEX] [dlstrip=REGEX] [keepfrag=REGEX]
#
#   <host>       subdomains match (example.com covers www.example.com)
#   keep=N       of the files one URL produced, keep only the first N
#   min=SIZE     delete produced files smaller than SIZE (500K, 1M, 2G)
#   id=REGEX     the path segment holding the item's identity, alone (/x/<id>)
#                or trailing a slug (/videos/<slug>-<id>)
#   also=HOSTS   more domains that are this same site, comma-separated
#   serial       crawl this site one URL at a time (avoids rate limits)
#   native       use yt-dlp's own downloader instead of aria2c (DASH streams)
#   probe=HOSTS  content servers TCP-checked before crawling
#   gdl=K=V,...  extra gallery-dl options for this site
#   key=PARAM    the query argument that names the item (watch?v=...)
#   playlist=SEG creator pages: <host>/<service>/<SEG>/<user> is a list of posts
#   postid=REGEX how a file of a post starts; group 1 is the post id
#   posturl=URL  the URL a post row is filed under; {host} {service} {user} {id}
#   itemurl=URL  the URL of one item, for probe mode; adds {slug}
#   tpl=TEMPLATE yt-dlp output template for this site
#   folder=REGEX file hosts: the collection folder name is capture group 1
#   engine=NAME  gdl/megadl/mdrs: the only engine for this host;
#                yt-dlp: yt-dlp first, gallery-dl only as a fallback
#   author=yt-dlp use yt-dlp's display name as the gallery-dl creator folder
#                when gallery-dl's metadata only exposes the handle
#   dest=NAME    destination from dredge.conf ([dest:NAME]); default otherwise
#   rewrite=REGEX=>URL  rewrite the URL before the engines see it (path/alias fix)
#   cookiedrop=NAMES  cookies a site must never receive (wildcards allowed)
#
# The global line (host "*") holds rules for every site:
#   strip=       query arguments that never identify an item
#   dlstrip=     the subset also stripped from the URL handed to the engines
#   keepfrag=    URL fragments that carry identity and must be kept

*                      strip=^(utm_.*|gclid|fbclid|igshid|msclkid|ad_sub|subid\d*|campaign_id|campaign|reff|ref_src|site_id|trx|promo|aff|usp|is_from_webapp|feature|ref|ch|t|time_continue|timestamp)$

example.com            keep=all  min=0
youtube.com            keep=all  min=0  key=v  also=youtu.be  native
